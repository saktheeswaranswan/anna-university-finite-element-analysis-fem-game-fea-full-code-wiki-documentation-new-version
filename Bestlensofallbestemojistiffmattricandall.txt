/* =========================================================
   UTILITIES
   ========================================================= */

boolean over(float x, float y, float w, float h) {
  return mouseX > x && mouseX < x + w && mouseY > y && mouseY < y + h;
}

boolean isPositiveIntString(String s) {
  if (s == null) return false;
  s = trim(s);
  if (s.length() == 0) return false;
  try {
    return Integer.parseInt(s) > 0;
  } catch (Exception e) {
    return false;
  }
}

boolean isFloatString(String s) {
  if (s == null) return false;
  s = trim(s);
  if (s.length() == 0) return false;
  try {
    Float.parseFloat(s);
    return true;
  } catch (Exception e) {
    return false;
  }
}

/* =========================================================
   MAIN APPLICATION (OOP CONTEXT CONTROLLER)
   ========================================================= */

class FEMApplication {
  FEMDataModel model;
  ScreenManager screenManager;
  Keypad keypad;
  
  String msg = "Enter GST size";
  
  FEMApplication() {
    model = new FEMDataModel();
    keypad = new Keypad(this);
    screenManager = new ScreenManager(this);
    keypad.startGST();
  }
  
  void render() {
    screenManager.render();
    keypad.draw();
    drawStatus();
  }
  
  void drawStatus() {
    fill(255);
    stroke(0);
    rect(20, height - 55, width - 40, 34, 8);
    fill(0);
    textSize(14);
    text(msg, width / 2, height - 38);
  }
  
  void handleMouse() {
    if (keypad.mousePressed()) return;
    if (screenManager.mousePressed()) return;
  }
  
  void handleDrag() {
    keypad.drag();
    screenManager.drag();
  }
  
  void handleRelease() {
    keypad.stopDrag();
    screenManager.stopDrag();
  }
}

/* =========================================================
   DATA MODEL DEFINITION
   ========================================================= */

class FEMDataModel {
  Element[] elements = new Element[MAX_E];
  GlobalMatrix G = new GlobalMatrix(MAX_G);
  
  int gstN = 0;
  int elementCount = 0;
  int active = 0;
  int assemblyActive = -1;
  
  boolean gstSet = false;
  boolean settingFinished = false;
  boolean assembling = false;
  boolean locked = false;
  
  FEMDataModel() {
    for (int i = 0; i < MAX_E; i++) {
      elements[i] = new Element();
    }
  }
  
  Element getActiveElement() {
    if (elementCount <= 0 || active < 0 || active >= elementCount) return null;
    return elements[active];
  }
  
  boolean allReady() {
    if (!gstSet || elementCount <= 0) return false;
    for (int i = 0; i < elementCount; i++) {
      if (!elements[i].complete()) return false;
    }
    return true;
  }
  
  boolean allAssembled() {
    for (int i = 0; i < elementCount; i++) {
      if (elements[i].readyForAssembly()) return false;
    }
    return true;
  }
  
  void invalidateAssemblyState() {
    settingFinished = false;
    assembling = false;
    locked = false;
    assemblyActive = -1;
    G.reset();
    for (int i = 0; i < elementCount; i++) {
      elements[i].resetAssemblyProgress();
    }
  }
}

/* =========================================================
   SCREEN VISUAL WORKSPACE MANAGER
   ========================================================= */

class ScreenManager {
  FEMApplication parent;
  int currentScreen = SCREEN_EDITOR;
  
  EditorView editorView;
  CompareView compareView;
  GSTLensView gstLensView;
  
  UIButton btnTab1, btnTab2, btnTab3;
  
  ScreenManager(FEMApplication p) {
    this.parent = p;
    editorView = new EditorView(parent);
    compareView = new CompareView(parent);
    gstLensView = new GSTLensView(parent);
    
    btnTab1 = new UIButton(20, 15, 140, 36, "1: EDITOR WORKSPACE");
    btnTab2 = new UIButton(170, 15, 140, 36, "2: CROSS COMPARE");
    btnTab3 = new UIButton(320, 15, 140, 36, "3: LENS VIEWER");
  }
  
  void render() {
    fill(235);
    noStroke();
    rect(0, 0, width, 65);
    stroke(180);
    line(0, 65, width, 65);
    
    btnTab1.draw(currentScreen == SCREEN_EDITOR);
    btnTab2.draw(currentScreen == SCREEN_COMPARE);
    btnTab3.draw(currentScreen == SCREEN_GSTVIEW);
    
    fill(50);
    textSize(20);
    textAlign(RIGHT, CENTER);
    text("FEM SYSTEM ANALYSIS RUNTIME", width - 40, 32);
    textAlign(CENTER, CENTER);
    
    pushMatrix();
    if (currentScreen == SCREEN_EDITOR) editorView.render();
    else if (currentScreen == SCREEN_COMPARE) compareView.render();
    else if (currentScreen == SCREEN_GSTVIEW) gstLensView.render();
    popMatrix();
  }
  
  boolean mousePressed() {
    if (btnTab1.hit()) { currentScreen = SCREEN_EDITOR; return true; }
    if (btnTab2.hit()) { currentScreen = SCREEN_COMPARE; return true; }
    if (btnTab3.hit()) { currentScreen = SCREEN_GSTVIEW; return true; }
    
    if (currentScreen == SCREEN_EDITOR) return editorView.mousePressed();
    if (currentScreen == SCREEN_COMPARE) return compareView.mousePressed();
    if (currentScreen == SCREEN_GSTVIEW) return gstLensView.mousePressed();
    return false;
  }
  
  void drag() {
    if (currentScreen == SCREEN_EDITOR) editorView.handleDrag();
    if (currentScreen == SCREEN_COMPARE) compareView.handleDrag();
    if (currentScreen == SCREEN_GSTVIEW) gstLensView.handleDrag();
  }
  
  void stopDrag() {
    if (currentScreen == SCREEN_EDITOR) editorView.handleRelease();
    if (currentScreen == SCREEN_COMPARE) compareView.handleRelease();
    if (currentScreen == SCREEN_GSTVIEW) gstLensView.handleRelease();
  }
}

/* =========================================================
   SCREEN 1: STANDARD WORKSPACE CORE INTERFACE
   ========================================================= */

class EditorView {
  FEMApplication p;
  ZoomControl localZoom, gstZoom;
  UIButton[] controlButtons;
  
  EditorView(FEMApplication p) {
    this.p = p;
    localZoom = new ZoomControl("LOCAL MATRIX", 1.0, 0.10, 10.0, 0.10);
    gstZoom = new ZoomControl("GLOBAL GST", 1.0, 0.10, 10.0, 0.10);
    
    controlButtons = new UIButton[] {
      new UIButton(40, 0, 110, 36, "GST"),
      new UIButton(160, 0, 110, 36, "ELEMENTS"),
      new UIButton(280, 0, 100, 36, "SIZE"),
      new UIButton(390, 0, 110, 36, "FINISH"),
      new UIButton(510, 0, 110, 36, "ASSEMBLE"),
      new UIButton(630, 0, 120, 36, "AUTO MAT"),
      new UIButton(760, 0, 120, 36, "AUTO DOF"),
      new UIButton(890, 0, 120, 36, "AUTO BAND")
    };
  }
  
  void render() {
    FEMDataModel m = p.model;
    Element e = m.getActiveElement();
    
    drawHeaderText(e);
    drawElementSelectors();
    
    float localCell = computeLocalCell(e);
    float globalCell = computeGlobalCell();
    
    if (e != null && e.size > 0) {
      e.drawLocal(40, 135, localCell, p.keypad, m.assembling && m.active == m.assemblyActive, localZoom.scale, m.active);
      float dofY = 145 + e.size * localCell * localZoom.scale + 45;
      e.drawDOF(40, dofY, localCell, p.keypad, m.assembling && m.active == m.assemblyActive, localZoom.scale);
    } else {
      fill(130);
      textSize(15);
      text(m.gstSet ? "Select an element and click 'SIZE' to configure properties." : "Configure Global System Stiffness Matrix dimensions to initialize.", width * 0.22, 250);
    }
    
    if (m.gstSet) {
      m.G.draw(width * 0.55, 135, globalCell, gstZoom.scale);
    }
    
    localZoom.draw(40, 85, e != null ? e.size : 0);
    gstZoom.draw(width * 0.55, 85, m.gstN);
    
    drawControlButtons();
  }
  
  void drawHeaderText(Element e) {
    FEMDataModel m = p.model;
    fill(0);
    textSize(15);
    textAlign(LEFT, CENTER);
    String meta = "System Nodes: " + (m.gstSet ? str(m.gstN) : "-") + " | Total Elements: " + (m.elementCount > 0 ? str(m.elementCount) : "-") + " | Engine Mode: " + (m.locked ? "LOCKED" : (m.settingFinished ? "STABLE SYSTEM" : "MUTABLE PIPELINE"));
    text(meta, 40, height - 235);
    
    String stateLabel = "Status: Idle Workspace Matrix Setup Context Engine";
    if (m.locked) stateLabel = "Status: CONFIGURATION LOCKED - DEPLOY ASSEMBLY ENGINE ROUTINES";
    else if (!m.gstSet) stateLabel = "Status: Global Stiffness Matrix Boundary Incomplete";
    else if (e == null) stateLabel = "Status: Select an Element Block to Map System Array Indices";
    else if (e.size == 0) stateLabel = "Status: Inputting Structural Boundary Constraints Profile Dimensions";
    else if (!e.matrixComplete()) stateLabel = "Status: populating Local Element Property Matrices Values Data";
    else if (!e.dofComplete()) stateLabel = "Status: Binding Local Boundary Degrees of Freedom Coordinates Transformations";
    else stateLabel = "Status: Element Configuration Context Confirmed Standard Profile Checked Matrix";
    text(stateLabel, 40, height - 210);
    textAlign(CENTER, CENTER);
  }
  
  void drawElementSelectors() {
    FEMDataModel m = p.model;
    for (int i = 0; i < MAX_E; i++) {
      float x = width * 0.55 + (i % 20) * 32;
      float y = height - 240 + (i / 20) * 26;
      int col = color(210);
      if (i >= m.elementCount) col = color(230, 230, 230, 150);
      else if (m.assembling && i == m.assemblyActive) col = color(255, 160, 50);
      else if (i == m.active) col = color(100, 180, 255);
      else if (m.elements[i].complete()) col = color(150, 235, 150);
      else if (m.elements[i].size > 0) col = color(235, 235, 180);
      
      fill(col);
      stroke(100);
      rect(x, y, 28, 22, 4);
      fill(0);
      textSize(10);
      text(i + 1, x + 14, y + 11);
    }
  }
  
  void drawControlButtons() {
    FEMDataModel m = p.model;
    float btnY = height - 110;
    Element e = m.getActiveElement();
    
    for (UIButton b : controlButtons) b.y = btnY;
    
    controlButtons[0].drawInner(true);
    controlButtons[1].drawInner(m.gstSet);
    controlButtons[2].drawInner(m.gstSet && e != null);
    controlButtons[3].drawInner(m.allReady());
    controlButtons[4].drawInner(m.settingFinished);
    
    boolean eValid = (e != null && e.size > 0);
    controlButtons[5].drawInner(eValid && !m.locked);
    controlButtons[6].drawInner(eValid && e.matrixComplete() && !m.locked);
    controlButtons[7].drawInner(eValid && e.matrixComplete() && !m.locked);
  }
  
  boolean mousePressed() {
    FEMDataModel m = p.model;
    if (localZoom.mousePressed() || gstZoom.mousePressed()) return true;
    
    float btnY = height - 110;
    for (UIButton b : controlButtons) {
      if (over(b.x, btnY, b.w, b.h)) {
        processButtonAction(b.text);
        return true;
      }
    }
    
    for (int i = 0; i < m.elementCount; i++) {
      float x = width * 0.55 + (i % 20) * 32;
      float y = height - 240 + (i / 20) * 26;
      if (over(x, y, 28, 22)) {
        m.active = i;
        p.msg = "Selected Element " + (m.active + 1) + ".";
        return true;
      }
    }
    
    Element e = m.getActiveElement();
    if (e == null || e.size <= 0) return false;
    
    float cell = computeLocalCell(e) * localZoom.scale;
    for (int r = 0; r < e.size; r++) {
      for (int c = 0; c < e.size; c++) {
        float x = 40 + c * cell;
        float y = 135 + r * cell;
        if (over(x, y, cell, cell)) {
          p.keypad.startMatrix(e, r, c);
          p.msg = "Enter matrix coefficient for Element " + (m.active + 1) + " at (" + (r + 1) + "," + (c + 1) + ").";
          return true;
        }
      }
    }
    
    float dofY = 145 + e.size * cell + 45;
    for (int i = 0; i < e.size; i++) {
      float x = 40 + i * cell;
      if (over(x, dofY, cell, cell)) {
        p.keypad.startDOF(e, i);
        p.msg = "Assign Global Boundary Degree of Freedom configuration reference index for local coordinate " + (i + 1) + ".";
        return true;
      }
    }
    return false;
  }
  
  void processButtonAction(String name) {
    FEMDataModel m = p.model;
    Element e = m.getActiveElement();
    
    if (m.locked && !name.equals("ASSEMBLE")) {
      p.msg = "Stiffness calculations locked. Execute structural assembly integration step.";
      return;
    }
    
    if (name.equals("GST")) {
      p.keypad.startGST();
    } else if (name.equals("ELEMENTS")) {
      if (!m.gstSet) { p.msg = "Global System Stiffness Matrix workspace boundaries undefined."; return; }
      p.keypad.startElementCount();
    } else if (name.equals("SIZE")) {
      if (e == null) { p.msg = "Target element instance selection array scope execution error."; return; }
      p.keypad.startSize(e);
    } else if (name.equals("FINISH")) {
      if (m.allReady()) {
        m.settingFinished = true; m.assembling = false; m.assemblyActive = -1; m.locked = true;
        p.keypad.mode = INPUT_NONE; p.msg = "WORKSPACE SYSTEM VERIFIED & LOCKED. READY FOR INTEGRATION STEP.";
      } else {
        p.msg = "Validation Error: Unmapped matrices entries exist within global collection scope.";
      }
    } else if (name.equals("ASSEMBLE")) {
      if (!m.settingFinished) { p.msg = "Pipeline error: Compile structural components data before step integration."; return; }
      m.assembling = true;
      executeAssemblyStep();
    } else if (name.equals("AUTO MAT") && e != null) {
      e.autoMatrix(); m.invalidateAssemblyState();
      p.keypad.startDOF(e, 0); p.msg = "Element data auto-generated. Map systemic nodes.";
    } else if (name.equals("AUTO DOF") && e != null && e.matrixComplete()) {
      e.autoDOF(m.gstN); m.invalidateAssemblyState();
      if (e.complete()) handleElementSetupSequence();
    } else if (name.equals("AUTO BAND") && e != null && e.matrixComplete()) {
      e.autoBandDOF(m.gstN, m.active + 1); m.invalidateAssemblyState();
      if (e.complete()) handleElementSetupSequence();
    }
  }
  
  void handleElementSetupSequence() {
    FEMDataModel m = p.model;
    if (m.active < m.elementCount - 1) {
      m.active++; p.keypad.startSize(m.elements[m.active]);
      p.msg = "Configuring element matrix component sequence item index profile: " + (m.active + 1);
    } else {
      p.msg = "All matrix arrays mapped. Validate structure calculations compilation using 'FINISH'.";
      p.keypad.mode = INPUT_NONE;
    }
  }
  
  void executeAssemblyStep() {
    FEMDataModel m = p.model;
    if (!m.settingFinished || m.elementCount == 0) { m.assembling = false; return; }
    
    for (int i = 0; i < m.elementCount; i++) {
      int idx = (m.assemblyActive + 1 + i) % m.elementCount;
      Element e = m.elements[idx];
      if (e.readyForAssembly()) {
        m.active = idx; m.assemblyActive = idx;
        if (e.stepOne(m.G)) {
          m.G.setHighlight(e.lastGI, e.lastGJ, e.lastValue, idx);
          p.msg = "Integration Component Logic: E" + (idx + 1) + " Matrix mapping indices data component updates global system [" + (e.lastGI + 1) + "," + (e.lastGJ + 1) + "] += " + nf(e.lastValue, 0, 2);
          if (e.done) p.msg += " | Structural integration vector segment completed for Element " + (idx + 1);
        }
        if (m.allAssembled()) { m.assembling = false; p.msg = "All Element Local Systems Compiled Into Global Structure Matrix Object Successfully."; }
        return;
      }
    }
    m.assembling = false;
    p.msg = "System Stiffness Architecture Analysis Complete.";
  }
  
  float computeLocalCell(Element e) {
    if (e == null || e.size <= 0) return 36;
    return constrain(min(40, (width * 0.35f) / e.size), 8, 40);
  }
  
  float computeGlobalCell() {
    FEMDataModel m = p.model;
    if (!m.gstSet || m.G.N <= 0) return 25;
    return constrain(min(26, (height * 0.52f) / m.G.N), 4, 26);
  }
  
  void handleDrag() { localZoom.drag(); gstZoom.drag(); }
  void handleRelease() { localZoom.stopDrag(); gstZoom.stopDrag(); }
}

/* =========================================================
   SCREEN 2: ALL MATRIX CROSS COMPARISON WORKSPACE (WITH LENS)

class CompareView {
  FEMApplication p;
  ZoomControl viewZoom;
  UIButton btnNext, btnPrev;
  
  int compRow = 0, compCol = 0;
  int curPage = 0;
  int itemsPerPage = 6;
  
  float lensD = 220;
  boolean enabledLens = true;
  
  CompareView(FEMApplication p) {
    this.parentWindowWorkContextAssign(p);
    viewZoom = new ZoomControl("VIEW MATRIX COMP", 0.85f, 0.15f, 4.0f, 0.05f);
    btnPrev = new UIButton(40, height - 110, 100, 36, "PREVIOUS");
    btnNext = new UIButton(160, height - 110, 100, 36, "NEXT PAGE");
  }
  
  void parentWindowWorkContextAssign(FEMApplication context) { this.p = context; }
  
  void render() {
    FEMDataModel m = p.model;
    viewZoom.draw(40, 80, m.elementCount);
    
    fill(30);
    textSize(18);
    textAlign(LEFT, CENTER);
    text("SYSTEM ARCHITECTURE ELEMENT MULTI-ARRAY FIELD TRACKER", 270, 110);
    text("Cross Tracking Vector Coordinate Targets: Row " + (compRow + 1) + " | Column " + (compCol + 1) + "  (Click cell space to map multi-array tracker references)", 40, height - 150);
    textAlign(CENTER, CENTER);
    
    btnPrev.drawInner(curPage > 0);
    btnNext.drawInner((curPage + 1) * itemsPerPage < m.elementCount);
    
    int startIdx = curPage * itemsPerPage;
    int endIdx = min(m.elementCount, startIdx + itemsPerPage);
    
    float spaceX = width * 0.31f;
    float spaceY = height * 0.31f;
    float baseCell = 36 * viewZoom.scale;
    
    Element hoveredElement = null;
    int hoveredElementId = -1;
    int hoveredRow = -1, hoveredCol = -1;
    float hoveredMatrixX = 0, hoveredMatrixY = 0;
    
    for (int i = startIdx; i < endIdx; i++) {
      int localPos = i - startIdx;
      float matrixX = 40 + (localPos % 3) * spaceX;
      float matrixY = 160 + (localPos / 3) * spaceY;
      
      drawCompareCellBlock(m.elements[i], matrixX, matrixY, i, baseCell);
      
      // Hitbox testing to find out if mouse coordinates track over an allocated cellular block
      Element e = m.elements[i];
      if (e != null && e.size > 0) {
        for (int r = 0; r < e.size; r++) {
          for (int c = 0; c < e.size; c++) {
            if (over(matrixX + c * baseCell, matrixY + r * baseCell, baseCell, baseCell)) {
              hoveredElement = e;
              hoveredElementId = i;
              hoveredRow = r;
              hoveredCol = c;
              hoveredMatrixX = matrixX;
              hoveredMatrixY = matrixY;
            }
          }
        }
      }
    }
    
    if (enabledLens && hoveredElement != null) {
      executeCompareLensPipeline(hoveredElement, hoveredElementId, hoveredRow, hoveredCol, baseCell, hoveredMatrixX, hoveredMatrixY);
    }
  }
  
  void drawCompareCellBlock(Element e, float x, float y, int id, float baseCell) {
    if (e == null || e.size <= 0) return;
    
    fill(50);
    textSize(13);
    textAlign(LEFT, CENTER);
    text("LOCAL ARRAY ELEMENT ID: " + (id + 1) + " (Bound Size: " + e.size + ")", x, y - 16);
    textAlign(CENTER, CENTER);
    
    for (int r = 0; r < e.size; r++) {
      for (int c = 0; c < e.size; c++) {
        float cx = x + c * baseCell;
        float cy = y + r * baseCell;
        
        boolean matchRow = (r == compRow);
        boolean matchCol = (c == compCol);
        
        if (matchRow && matchCol) fill(255, 100, 100);
        else if (matchRow || matchCol) fill(255, 210, 210);
        else fill(e.keSet[r][c] ? 240 : 255);
        
        stroke(120);
        rect(cx, cy, baseCell, baseCell);
        
        if (e.keSet[r][c] && baseCell > 14) {
          fill(0);
          textSize(constrain(baseCell * 0.28f, 6, 12));
          text(nf(e.ke[r][c], 0, 1), cx + baseCell / 2, cy + baseCell / 2);
        }
        
        if (matchRow && matchCol) {
          fill(0);
          textSize(constrain(baseCell * 0.6f, 12, 24));
          text("☝", cx + baseCell / 2, cy + baseCell / 2);
        }
      }
    }
  }
  
  void executeCompareLensPipeline(Element e, int elementId, int targetR, int targetC, float cell, float sx, float sy) {
    float lx = mouseX;
    float ly = mouseY;
    
    pushMatrix();
    pushStyle();
    
    // -----------------------------------------------------------------
    // FLOATING HEADS-UP DISPLAY PANEL CALLOUT FOR COMPARE ENGINE
    // -----------------------------------------------------------------
    float panelW = 280;
    float panelH = 190;
    
    float panelX = lx + lensD/2 + 20;
    if (panelX + panelW > width - 30) {
      panelX = lx - lensD/2 - panelW - 20;
    }
    float panelY = constrain(ly - panelH/2, 130, height - panelH - 100);
    
    // Draw HUD Panel Background
    fill(25, 30, 40, 240);
    stroke(240, 120, 80); // Alternate tracking orange/coral line signature color for matrix comparisons
    strokeWeight(1.5f);
    rect(panelX, panelY, panelW, panelH, 8);
    strokeWeight(1.0f);
    
    // Header
    fill(240, 120, 80);
    textSize(11);
    textAlign(LEFT, CENTER);
    text("COMPARE LENS MATRIX ANALYSIS HUD", panelX + 15, panelY + 18);
    
    stroke(60, 70, 90);
    line(panelX + 15, panelY + 32, panelX + panelW - 15, panelY + 32);
    
    // Display Parameters Data fields
    fill(220);
    textSize(12);
    text("Active Element Source:", panelX + 15, panelY + 52);
    fill(255);
    text("Element ID Matrix Profile " + (elementId + 1), panelX + 160, panelY + 52);
    
    fill(220);
    text("Local Array Position:", panelX + 15, panelY + 76);
    fill(255);
    text("Row " + (targetR + 1) + " , Col " + (targetC + 1), panelX + 160, panelY + 76);
    
    fill(220);
    text("Full Floating Precision:", panelX + 15, panelY + 100);
    float absoluteVal = e.ke[targetR][targetC];
    if (!e.keSet[targetR][targetC]) {
      fill(130, 140, 150);
      text("Unallocated Void Scalar", panelX + 160, panelY + 100);
    } else {
      fill(100, 240, 140);
      text(nf(absoluteVal, 0, 4), panelX + 160, panelY + 100);
    }
    
    fill(220);
    text("Boundary Node Routing:", panelX + 15, panelY + 124);
    fill(255);
    String boundLabel = e.dofSet[targetR] ? "g" + (e.dof[targetR] + 1) : "None";
    text("DOF Mapping Vector: " + boundLabel, panelX + 160, panelY + 124);
    
    // Status Bar inside floating panel
    stroke(60, 70, 90);
    line(panelX + 15, panelY + 144, panelX + panelW - 15, panelY + 144);
    
    fill(160, 180, 200);
    textSize(10);
    text("PIPELINE ENGINE: CELL RESOLUTION MAGNIFIED", panelX + 15, panelY + 164);

/* =========================================================
   ELEMENT MATRIX ENTITY SCHEMATIC DEFINITIONS
   ========================================================= */

class Element {
  int size = 0;
  float[][] ke;
  boolean[][] keSet;
  
  int[] dof;
  boolean[] dofSet;
  
  int asmR = 0, asmC = 0;
  boolean done = false;
  int lastR = -1, lastC = -1, lastGI = -1, lastGJ = -1;
  float lastValue = 0;
  
  void init(int n) {
    size = n;
    ke = new float[n][n];
    keSet = new boolean[n][n];
    dof = new int[n];
    dofSet = new boolean[n];
    resetAssemblyProgress();
  }
  
  void resetAll() {
    size = 0; ke = null; keSet = null; dof = null; dofSet = null;
    resetAssemblyProgress();
  }
  
  void resetAssemblyProgress() {
    asmR = 0; asmC = 0; done = false;
    lastR = -1; lastC = -1; lastGI = -1; lastGJ = -1; lastValue = 0;
  }
  
  void autoMatrix() {
    if (size <= 0) return;
    for (int r = 0; r < size; r++) {
      for (int c = 0; c < size; c++) {
        ke[r][c] = random(-500.0f, 1000.0f); // Provides full realistic float ranges
        keSet[r][c] = true;
      }
    }
    resetAssemblyProgress();
  }
  
  void autoDOF(int gstN) {
    if (size <= 0 || gstN < size) return;
    ArrayList<Integer> pool = new ArrayList<Integer>();
    for (int i = 0; i < gstN; i++) pool.add(i + 1);
    for (int i = 0; i < size; i++) {
      int rIdx = int(random(pool.size()));
      dof[i] = pool.remove(rIdx) - 1;
      dofSet[i] = true;
    }
    resetAssemblyProgress();
  }
  
  void autoBandDOF(int gstN, int startDOF) {
    if (size <= 0 || gstN < size) return;
    int start = constrain(startDOF, 1, gstN - size + 1);
    for (int i = 0; i < size; i++) {
      dof[i] = (start - 1) + i;
      dofSet[i] = true;
    }
    resetAssemblyProgress();
  }
  
  boolean matrixComplete() {
    if (size == 0) return false;
    for (int r = 0; r < size; r++) {
      for (int c = 0; c < size; c++) if (!keSet[r][c]) return false;
    }
    return true;
  }
  
  boolean dofComplete() {
    if (size == 0) return false;
    for (int i = 0; i < size; i++) if (!dofSet[i]) return false;
    return true;
  }
  
  boolean complete() { return size > 0 && matrixComplete() && dofComplete(); }
  boolean readyForAssembly() { return complete() && !done; }
  
  boolean stepOne(GlobalMatrix G) {
    if (!readyForAssembly()) return false;
    lastR = asmR; lastC = asmC;
    lastGI = dof[asmR]; lastGJ = dof[asmC];
    lastValue = ke[asmR][asmC];
    G.add(lastGI, lastGJ, lastValue);
    
    asmC++;
    if (asmC >= size) { asmC = 0; asmR++; }
    if (asmR >= size) { done = true; }
    return true;
  }
  
  void drawLocal(float x, float y, float cell, Keypad kp, boolean hiAsm, float zoom, int actId) {
    pushStyle();
    fill(0); textSize(14); textAlign(LEFT, CENTER);
    text("LOCAL STIFFNESS COMPONENT: ELEMENT " + (actId + 1), x, y - 20);
    textAlign(CENTER, CENTER);
    
    float dCell = cell * zoom;
    textSize(constrain(dCell * 0.32f, 6, 13));
    
    for (int r = 0; r < size; r++) {
      for (int c = 0; c < size; c++) {
        boolean sel = (kp.mode == INPUT_MATRIX && kp.el == this && kp.r == r && kp.c == c);
        if (hiAsm && r == lastR && c == lastC) fill(255, 150, 150);
        else if (sel) fill(255, 230, 140);
        else fill(keSet[r][c] ? 240 : 255);
        
        stroke(50);
        rect(x + c * dCell, y + r * dCell, dCell, dCell);
        
        fill(0);
        String t = "";
        if (keSet[r][c]) t = nf(ke[r][c], 0, 1);
        if (sel && kp.val.length() > 0) t = kp.val;
        text(t, x + c * dCell + dCell / 2, y + r * dCell + dCell / 2);
      }
    }
    popStyle();
  }
  
  void drawDOF(float x, float y, float cell, Keypad kp, boolean hiAsm, float zoom) {
    pushStyle();
    fill(0); textSize(13); textAlign(LEFT, CENTER);
    text("SYSTEM NODE MAP TRANSLATIONS (Degrees of Freedom Boundary Coordinates)", x, y - 16);
    textAlign(CENTER, CENTER);
    
    float dCell = cell * zoom;
    textSize(constrain(dCell * 0.34f, 6, 13));
    
    for (int i = 0; i < size; i++) {
      boolean sel = (kp.mode == INPUT_DOF && kp.index == i);
      if (hiAsm && (i == lastR || i == lastC)) fill(255, 150, 150);
      else if (sel) fill(255, 230, 140);
      else fill(dofSet[i] ? 240 : 255);
      
      stroke(50);
      rect(x + i * dCell, y, dCell, dCell);
      
      fill(0);
      String t = "";
      if (dofSet[i]) t = "g" + (dof[i] + 1);
      if (sel && kp.val.length() > 0) t = kp.val;
      text(t, x + i * dCell + dCell / 2, y + dCell / 2);
    }
    popStyle();
  }
}

/* =========================================================
   GLOBAL ARCHITECTURE RUNTIME REGISTER MATRIX
   ========================================================= */

class GlobalMatrix {
  int N = 0;
  float[][] K;
  int[][] records;
  
  int hiI = -1, hiJ = -1, hiElem = -1;
  float hiV = 0;
  
  GlobalMatrix(int max) {
    K = new float[max][max];
    records = new int[max][max];
  }
  
  void setSize(int n) { this.N = n; reset(); }
  
  void reset() {
    for (int i = 0; i < K.length; i++) {
      for (int j = 0; j < K.length; j++) { K[i][j] = 0; records[i][j] = 0; }
    }
    clearHighlight();
  }
  
  void clearHighlight() { hiI = -1; hiJ = -1; hiElem = -1; hiV = 0; }
  void setHighlight(int I, int J, float v, int elem) { hiI = I; hiJ = J; hiV = v; hiElem = elem; }
  
  void add(int I, int J, float val) {
    if (I >= 0 && J >= 0 && I < N && J < N) { K[I][J] += val; records[I][J]++; }
  }
  
  void draw(float x, float y, float cell, float zoom) {
    pushStyle();
    fill(0); textSize(14); textAlign(LEFT, CENTER);
    text("GLOBAL GLOBAL STIFFNESS SYSTEM ARRAY (" + N + " x " + N + ")", x, y - 20);
    textAlign(CENTER, CENTER);
    
    float dCell = cell * zoom;
    textSize(constrain(dCell * 0.32f, 5, 12));
    
    for (int r = 0; r < N; r++) {
      for (int c = 0; c < N; c++) {
        if (r == hiI && c == hiJ) fill(255, 150, 150);
        else if (records[r][c] > 0) fill(160, 220, 180);
        else fill(255);
        
        stroke(160);
        rect(x + c * dCell, y + r * dCell, dCell, dCell);
        
        if (dCell > 10 || records[r][c] > 0) {
          fill(0);
          text(nf(K[r][c], 0, 1), x + c * dCell + dCell / 2, y + r * dCell + dCell / 2);
        }
      }
    }
    popStyle();
  }
}

  boolean mousePressed() {
    if (mode == INPUT_NONE) return false;
    if (over(x, y, w, 32)) { dragging = true; offX = mouseX - x; offY = mouseY - y; return true; }
    
    for (int i = 0; i < gridLayout.length; i++) {
      float bx = x + 12 + (i % 3) * 76;
      float by = y + 42 + (i / 3) * 42;
      if (over(bx, by, 70, 36)) {
        processKeyStroke(gridLayout[i]);
        return true;
      }
    }
    return over(x, y, w, h);
  }
  
  void processKeyStroke(String key) {
    if (key.equals("C")) val = "";
    else if (key.equals("⌫")) { if (val.length() > 0) val = val.substring(0, val.length() - 1); }
    else if (key.equals("OK")) commit();
    else {
      if (mode == INPUT_MATRIX && (key.equals(".") || key.equals("-"))) val += key;
      else if (Character.isDigit(key.charAt(0))) val += key;
    }
  }
  
  void commit() {
    FEMDataModel m = app.model;
    if (val.length() == 0) return;
    
    if (mode == INPUT_GST) {
      if (!isPositiveIntString(val)) return;
      m.gstN = constrain(int(val), 1, MAX_G);
      m.gstSet = true; m.G.setSize(m.gstN);
      m.invalidateAssemblyState(); m.elementCount = 0;
      mode = INPUT_ECOUNT; app.msg = "Global node boundaries set. Enter structural element count.";
      val = ""; return;
    }
    if (mode == INPUT_ECOUNT) {
      if (!isPositiveIntString(val)) return;
      m.elementCount = constrain(int(val), 1, MAX_E);
      m.invalidateAssemblyState();
      for (int i = 0; i < MAX_E; i++) m.elements[i].resetAll();
      m.active = 0; mode = INPUT_SIZE; el = m.elements[0];
      app.msg = "Stiffness elements structured. Setup parameters array boundaries for Element 1.";
      val = ""; return;
    }
    if (mode == INPUT_SIZE) {
      if (!isPositiveIntString(val)) return;
      int s = constrain(int(val), 1, m.gstN);
      el.init(s); m.invalidateAssemblyState();
      startMatrix(el, 0, 0);
      app.msg = "Internal bounds loaded. Input processing cell constants.";
      return;
    }
    if (mode == INPUT_MATRIX) {
      if (!isFloatString(val)) return;
      el.ke[r][c] = float(val); el.keSet[r][c] = true;
      m.invalidateAssemblyState();
      int total = el.size * el.size;
      int next = r * el.size + c + 1;
      if (next < total) { startMatrix(el, next / el.size, next % el.size); }
      else { startDOF(el, 0); app.msg = "Cell configurations complete. Bind systemic target boundary vector paths."; }
      return;
    }
    if (mode == INPUT_DOF) {
      if (!isPositiveIntString(val)) return;
      int d = constrain(int(val), 1, m.gstN);
      el.dof[index] = d - 1; el.dofSet[index] = true;
      m.invalidateAssemblyState();
      if (index + 1 < el.size) { startDOF(el, index + 1); }
      else {
        el.resetAssemblyProgress();
        if (m.active < m.elementCount - 1) {
          m.active++; el = m.elements[m.active]; mode = INPUT_SIZE;
          app.msg = "Element validation matrix structured. Define dimensional scale for Component: " + (m.active + 1);
        } else {
          app.msg = "Stiffness compilation matrices elements verification checked. Execute layout compiling with 'FINISH'.";
          mode = INPUT_NONE;
        }
      }
    }
    val = "";
  }
  
  void drag() { if (dragging) { x = mouseX - offX; y = mouseY - offY; } }
  void stopDrag() { dragging = false; }
}
