final int MAX_E = 100;
final int MAX_G = 100;

final int SCREEN_EDITOR  = 0;
final int SCREEN_COMPARE = 1;
final int SCREEN_GSTVIEW = 2;

final int INPUT_NONE   = 0;
final int INPUT_GST    = 1;
final int INPUT_ECOUNT = 2;
final int INPUT_SIZE   = 3;
final int INPUT_MATRIX = 4;
final int INPUT_DOF    = 5;

FEMApplication app;

void setup() {
  fullScreen();
  smooth(8);
  textAlign(CENTER, CENTER);
  rectMode(CORNER);
  app = new FEMApplication();
}

void draw() {
  background(245);
  app.render();
}

void mousePressed()  { app.handleMouse(); }
void mouseDragged()   { app.handleDrag(); }
void mouseReleased()  { app.handleRelease(); }

/* =========================================================
   UTILITIES
   ========================================================= */

boolean over(float x, float y, float w, float h) {
  return mouseX > x && mouseX < x + w && mouseY > y && mouseY < y + h;
}

boolean isPositiveIntString(String s) {
  if (s == null) return false;
  s = trim(s);
  if (s.length() == 0) return false;
  try {
    return Integer.parseInt(s) > 0;
  } catch (Exception e) {
    return false;
  }
}

boolean isFloatString(String s) {
  if (s == null) return false;
  s = trim(s);
  if (s.length() == 0) return false;
  try {
    Float.parseFloat(s);
    return true;
  } catch (Exception e) {
    return false;
  }
}

String dofLabel(Element e, int idx) {
  if (e == null || idx < 0 || idx >= e.size || !e.dofSet[idx]) return "None";
  return "g" + (e.dof[idx] + 1);
}

/* =========================================================
   MAIN APPLICATION
   ========================================================= */

class FEMApplication {
  FEMDataModel model;
  ScreenManager screenManager;
  Keypad keypad;
  String msg = "Enter GST size";

  FEMApplication() {
    model = new FEMDataModel();
    keypad = new Keypad(this);
    screenManager = new ScreenManager(this);
    keypad.startGST();
  }

  void render() {
    screenManager.render();
    keypad.draw();
    drawStatus();
  }

  void drawStatus() {
    fill(255);
    stroke(0);
    rect(20, height - 55, width - 40, 34, 8);
    fill(0);
    textSize(14);
    text(msg, width / 2, height - 38);
  }

  void handleMouse() {
    if (keypad.mousePressed()) return;
    if (screenManager.mousePressed()) return;
  }

  void handleDrag() {
    keypad.drag();
    screenManager.drag();
  }

  void handleRelease() {
    keypad.stopDrag();
    screenManager.stopDrag();
  }
}

/* =========================================================
   DATA MODEL
   ========================================================= */

class FEMDataModel {
  Element[] elements = new Element[MAX_E];
  GlobalMatrix G = new GlobalMatrix(MAX_G);

  int gstN = 0;
  int elementCount = 0;
  int active = 0;
  int assemblyActive = -1;

  boolean gstSet = false;
  boolean settingFinished = false;
  boolean assembling = false;
  boolean locked = false;

  FEMDataModel() {
    for (int i = 0; i < MAX_E; i++) elements[i] = new Element();
  }

  Element getActiveElement() {
    if (elementCount <= 0 || active < 0 || active >= elementCount) return null;
    return elements[active];
  }

  boolean allReady() {
    if (!gstSet || elementCount <= 0) return false;
    for (int i = 0; i < elementCount; i++) {
      if (!elements[i].complete()) return false;
    }
    return true;
  }

  boolean allAssembled() {
    for (int i = 0; i < elementCount; i++) {
      if (elements[i].readyForAssembly()) return false;
    }
    return true;
  }

  void invalidateAssemblyState() {
    settingFinished = false;
    assembling = false;
    locked = false;
    assemblyActive = -1;
    G.reset();
    for (int i = 0; i < elementCount; i++) elements[i].resetAssemblyProgress();
  }
}

/* =========================================================
   SCREEN MANAGER
   ========================================================= */

class ScreenManager {
  FEMApplication parent;
  int currentScreen = SCREEN_EDITOR;

  EditorView editorView;
  CompareView compareView;
  GSTLensView gstLensView;

  UIButton btnTab1, btnTab2, btnTab3;

  ScreenManager(FEMApplication p) {
    this.parent = p;
    editorView = new EditorView(parent);
    compareView = new CompareView(parent);
    gstLensView = new GSTLensView(parent);

    btnTab1 = new UIButton(20, 15, 140, 36, "1: EDITOR WORKSPACE");
    btnTab2 = new UIButton(170, 15, 140, 36, "2: CROSS COMPARE");
    btnTab3 = new UIButton(320, 15, 140, 36, "3: LENS VIEWER");
  }

  void render() {
    fill(235);
    noStroke();
    rect(0, 0, width, 65);
    stroke(180);
    line(0, 65, width, 65);

    btnTab1.draw(currentScreen == SCREEN_EDITOR);
    btnTab2.draw(currentScreen == SCREEN_COMPARE);
    btnTab3.draw(currentScreen == SCREEN_GSTVIEW);

    fill(50);
    textSize(20);
    textAlign(RIGHT, CENTER);
    text("FEM SYSTEM ANALYSIS RUNTIME", width - 40, 32);
    textAlign(CENTER, CENTER);

    pushMatrix();
    if (currentScreen == SCREEN_EDITOR)      editorView.render();
    else if (currentScreen == SCREEN_COMPARE) compareView.render();
    else if (currentScreen == SCREEN_GSTVIEW) gstLensView.render();
    popMatrix();
  }

  boolean mousePressed() {
    if (btnTab1.hit()) {
      currentScreen = SCREEN_EDITOR;
      return true;
    }
    if (btnTab2.hit()) {
      currentScreen = SCREEN_COMPARE;
      return true;
    }
    if (btnTab3.hit()) {
      currentScreen = SCREEN_GSTVIEW;
      return true;
    }

    if (currentScreen == SCREEN_EDITOR)  return editorView.mousePressed();
    if (currentScreen == SCREEN_COMPARE) return compareView.mousePressed();
    if (currentScreen == SCREEN_GSTVIEW) return gstLensView.mousePressed();
    return false;
  }

  void drag() {
    if (currentScreen == SCREEN_EDITOR)  editorView.handleDrag();
    if (currentScreen == SCREEN_COMPARE) compareView.handleDrag();
    if (currentScreen == SCREEN_GSTVIEW) gstLensView.handleDrag();
  }

  void stopDrag() {
    if (currentScreen == SCREEN_EDITOR)  editorView.handleRelease();
    if (currentScreen == SCREEN_COMPARE) compareView.handleRelease();
    if (currentScreen == SCREEN_GSTVIEW) gstLensView.handleRelease();
  }
}

class EditorView {
  FEMApplication p;
  ZoomControl localZoom, gstZoom;
  UIButton[] controlButtons;

  EditorView(FEMApplication p) {
    this.p = p;
    localZoom = new ZoomControl("LOCAL MATRIX", 1.0, 0.10, 10.0, 0.10);
    gstZoom   = new ZoomControl("GLOBAL GST",   1.0, 0.10, 10.0, 0.10);

    controlButtons = new UIButton[] {
      new UIButton( 40, 0, 110, 36, "GST"),
      new UIButton(160, 0, 110, 36, "ELEMENTS"),
      new UIButton(280, 0, 100, 36, "SIZE"),
      new UIButton(390, 0, 110, 36, "FINISH"),
      new UIButton(510, 0, 110, 36, "ASSEMBLE"),
      new UIButton(630, 0, 120, 36, "AUTO MAT"),
      new UIButton(760, 0, 120, 36, "AUTO DOF"),
      new UIButton(890, 0, 120, 36, "AUTO BAND")
    };
  }

  void render() {
    FEMDataModel m = p.model;
    Element e = m.getActiveElement();

    drawHeaderText(e);
    drawElementSelectors();

    float localCell = computeLocalCell(e);
    float globalCell = computeGlobalCell();

    if (e != null && e.size > 0) {
      e.drawLocal(40, 135, localCell, p.keypad, m.assembling && m.active == m.assemblyActive, localZoom.scale, m.active);
      float dofY = 145 + e.size * localCell * localZoom.scale + 45;
      e.drawDOF(40, dofY, localCell, p.keypad, m.assembling && m.active == m.assemblyActive, localZoom.scale);
    } else {
      fill(130);
      textSize(15);
      text(m.gstSet ? "Select an element and click 'SIZE' to configure properties."
                    : "Configure Global System Stiffness Matrix dimensions to initialize.",
           width * 0.22, 250);
    }

    if (m.gstSet) m.G.draw(width * 0.55, 135, globalCell, gstZoom.scale);

    localZoom.draw(40, 85, e != null ? e.size : 0);
    gstZoom.draw(width * 0.55, 85, m.gstN);

    drawControlButtons();
  }

  void drawHeaderText(Element e) {
    FEMDataModel m = p.model;
    fill(0);
    textSize(15);
    textAlign(LEFT, CENTER);

    String meta = "System Nodes: " + (m.gstSet ? str(m.gstN) : "-") +
      " | Total Elements: " + (m.elementCount > 0 ? str(m.elementCount) : "-") +
      " | Engine Mode: " + (m.locked ? "LOCKED" : (m.settingFinished ? "STABLE SYSTEM" : "MUTABLE PIPELINE"));
    text(meta, 40, height - 235);

    String stateLabel = "Status: Idle Workspace Matrix Setup Context Engine";
    if (m.locked)                 stateLabel = "Status: CONFIGURATION LOCKED - DEPLOY ASSEMBLY ENGINE ROUTINES";
    else if (!m.gstSet)           stateLabel = "Status: Global Stiffness Matrix Boundary Incomplete";
    else if (e == null)           stateLabel = "Status: Select an Element Block to Map System Array Indices";
    else if (e.size == 0)         stateLabel = "Status: Inputting Structural Boundary Constraints Profile Dimensions";
    else if (!e.matrixComplete()) stateLabel = "Status: Populating Local Element Property Matrices Values Data";
    else if (!e.dofComplete())    stateLabel = "Status: Binding Local Boundary Degrees of Freedom Coordinates Transformations";
    else                          stateLabel = "Status: Element Configuration Context Confirmed Standard Profile Checked Matrix";

    text(stateLabel, 40, height - 210);
    textAlign(CENTER, CENTER);
  }

  void drawElementSelectors() {
    FEMDataModel m = p.model;
    for (int i = 0; i < MAX_E; i++) {
      float x = width * 0.55 + (i % 20) * 32;
      float y = height - 240 + (i / 20) * 26;

      int col = color(210);
      if      (i >= m.elementCount)                   col = color(230, 230, 230, 150);
      else if (m.assembling && i == m.assemblyActive) col = color(255, 160, 50);
      else if (i == m.active)                         col = color(100, 180, 255);
      else if (m.elements[i].complete())              col = color(150, 235, 150);
      else if (m.elements[i].size > 0)                col = color(235, 235, 180);

      fill(col);
      stroke(100);
      rect(x, y, 28, 22, 4);
      fill(0);
      textSize(10);
      text(i + 1, x + 14, y + 11);
    }
  }

  void drawControlButtons() {
    FEMDataModel m = p.model;
    float btnY = height - 110;
    Element e = m.getActiveElement();

    for (UIButton b : controlButtons) b.y = btnY;

    controlButtons[0].drawInner(true);
    controlButtons[1].drawInner(m.gstSet);
    controlButtons[2].drawInner(m.gstSet && e != null);
    controlButtons[3].drawInner(m.allReady());
    controlButtons[4].drawInner(m.settingFinished);

    boolean eValid = (e != null && e.size > 0);
    controlButtons[5].drawInner(eValid && !m.locked);
    controlButtons[6].drawInner(eValid && e.matrixComplete() && !m.locked);
    controlButtons[7].drawInner(eValid && e.matrixComplete() && !m.locked);
  }

  boolean mousePressed() {
    FEMDataModel m = p.model;

    if (localZoom.mousePressed() || gstZoom.mousePressed()) return true;

    float btnY = height - 110;
    for (UIButton b : controlButtons) {
      if (over(b.x, btnY, b.w, b.h)) {
        processButtonAction(b.text);
        return true;
      }
    }

    for (int i = 0; i < m.elementCount; i++) {
      float x = width * 0.55 + (i % 20) * 32;
      float y = height - 240 + (i / 20) * 26;
      if (over(x, y, 28, 22)) {
        m.active = i;
        p.msg = "Selected Element " + (m.active + 1) + ".";
        return true;
      }
    }

    Element e = m.getActiveElement();
    if (e == null || e.size <= 0) return false;

    float cell = computeLocalCell(e) * localZoom.scale;
    for (int r = 0; r < e.size; r++) {
      for (int c = 0; c < e.size; c++) {
        float x = 40 + c * cell;
        float y = 135 + r * cell;
        if (over(x, y, cell, cell)) {
          p.keypad.startMatrix(e, r, c);
          p.msg = "Enter matrix coefficient for Element " + (m.active + 1) + " at (" + (r + 1) + "," + (c + 1) + ").";
          return true;
        }
      }
    }

    float dofY = 145 + e.size * cell + 45;
    for (int i = 0; i < e.size; i++) {
      if (over(40 + i * cell, dofY, cell, cell)) {
        p.keypad.startDOF(e, i);
        p.msg = "Assign Global DOF reference index for local coordinate " + (i + 1) + ".";
        return true;
      }
    }

    return false;
  }

  void processButtonAction(String name) {
    FEMDataModel m = p.model;
    Element e = m.getActiveElement();

    if (m.locked && !name.equals("ASSEMBLE")) {
      p.msg = "Stiffness calculations locked. Execute structural assembly integration step.";
      return;
    }

    if (name.equals("GST")) {
      p.keypad.startGST();
    } else if (name.equals("ELEMENTS")) {
      if (!m.gstSet) {
        p.msg = "Global System Stiffness Matrix workspace boundaries undefined.";
        return;
      }
      p.keypad.startElementCount();
    } else if (name.equals("SIZE")) {
      if (e == null) {
        p.msg = "Target element instance selection array scope execution error.";
        return;
      }
      p.keypad.startSize(e);
    } else if (name.equals("FINISH")) {
      if (m.allReady()) {
        m.settingFinished = true;
        m.assembling = false;
        m.assemblyActive = -1;
        m.locked = true;
        p.keypad.mode = INPUT_NONE;
        p.msg = "WORKSPACE SYSTEM VERIFIED & LOCKED. READY FOR INTEGRATION STEP.";
      } else {
        p.msg = "Validation Error: Unmapped matrices entries exist within global collection scope.";
      }
    } else if (name.equals("ASSEMBLE")) {
      if (!m.settingFinished) {
        p.msg = "Pipeline error: Compile structural components data before step integration.";
        return;
      }
      m.assembling = true;
      executeAssemblyStep();
    } else if (name.equals("AUTO MAT") && e != null) {
      e.autoMatrix();
      m.invalidateAssemblyState();
      p.keypad.startDOF(e, 0);
      p.msg = "Element data auto-generated. Map systemic nodes.";
    } else if (name.equals("AUTO DOF") && e != null && e.matrixComplete()) {
      e.autoDOF(m.gstN);
      m.invalidateAssemblyState();
      if (e.complete()) handleElementSetupSequence();
    } else if (name.equals("AUTO BAND") && e != null && e.matrixComplete()) {
      e.autoBandDOF(m.gstN, m.active + 1);
      m.invalidateAssemblyState();
      if (e.complete()) handleElementSetupSequence();
    }
  }

  void handleElementSetupSequence() {
    FEMDataModel m = p.model;
    if (m.active < m.elementCount - 1) {
      m.active++;
      p.keypad.startSize(m.elements[m.active]);
      p.msg = "Configuring element matrix component sequence item index profile: " + (m.active + 1);
    } else {
      p.msg = "All matrix arrays mapped. Validate structure calculations compilation using 'FINISH'.";
      p.keypad.mode = INPUT_NONE;
    }
  }

  void executeAssemblyStep() {
    FEMDataModel m = p.model;
    if (!m.settingFinished || m.elementCount == 0) {
      m.assembling = false;
      return;
    }

    for (int i = 0; i < m.elementCount; i++) {
      int idx = (m.assemblyActive + 1 + i) % m.elementCount;
      Element e = m.elements[idx];

      if (e.readyForAssembly()) {
        m.active = idx;
        m.assemblyActive = idx;

        if (e.stepOne(m.G)) {
          m.G.setHighlight(e.lastGI, e.lastGJ, e.lastValue, idx);
          p.msg = "Integration: E" + (idx + 1) + " maps to GST [" + (e.lastGI + 1) + "," + (e.lastGJ + 1) + "] += " + nf(e.lastValue, 0, 2);
          if (e.done) p.msg += " | Element " + (idx + 1) + " fully assembled.";
        }

        if (m.allAssembled()) {
          m.assembling = false;
          p.msg = "All Element Local Systems Compiled Into Global Structure Matrix Successfully.";
        }
        return;
      }
    }

    m.assembling = false;
    p.msg = "System Stiffness Architecture Analysis Complete.";
  }

  float computeLocalCell(Element e) {
    if (e == null || e.size <= 0) return 36;
    return constrain(min(40, (width * 0.35f) / e.size), 8, 40);
  }

  float computeGlobalCell() {
    FEMDataModel m = p.model;
    if (!m.gstSet || m.G.N <= 0) return 25;
    return constrain(min(26, (height * 0.52f) / m.G.N), 4, 26);
  }

  void handleDrag()    { localZoom.drag(); gstZoom.drag(); }
  void handleRelease() { localZoom.stopDrag(); gstZoom.stopDrag(); }
}

/* =========================================================
   SCREEN 2: CROSS COMPARE
   Row and column of every contributing matrix
   Includes next/prev pair focus buttons
   Works before and after assembly
   ========================================================= */

class CompareView {
  FEMApplication p;
  ZoomControl viewZoom;
  UIButton btnPrevPage, btnNextPage, btnPrevPair, btnNextPair, btnToggleLens;

  int compRow = 0, compCol = 0;
  int curPage = 0;
  int itemsPerPage = 6;

  float lensD = 220;
  boolean enabledLens = true;

  final int blinkPeriod = 26;
  final int blinkOn = 13;

  int activeContributorIndex = 0;

  CompareView(FEMApplication p) {
    this.p = p;
    viewZoom = new ZoomControl("VIEW MATRIX COMP", 0.85f, 0.15f, 4.0f, 0.05f);
    btnPrevPage   = new UIButton( 40, height - 110, 100, 36, "PREVIOUS PAGE");
    btnNextPage   = new UIButton(160, height - 110, 100, 36, "NEXT PAGE");
    btnPrevPair   = new UIButton(310, height - 110, 100, 36, "PREV PAIR");
    btnNextPair   = new UIButton(420, height - 110, 100, 36, "NEXT PAIR");
    btnToggleLens = new UIButton(530, height - 110, 100, 36, "LENS ON/OFF");
  }

  int countContributors() {
    FEMDataModel m = p.model;
    int count = 0;
    for (int i = 0; i < m.elementCount; i++) {
      Element e = m.elements[i];
      if (e == null || e.size <= 0) continue;
      if (compRow < e.size && compCol < e.size && e.keSet[compRow][compCol]) count++;
    }
    return count;
  }

  ArrayList<Integer> getContributorIDs() {
    FEMDataModel m = p.model;
    ArrayList<Integer> ids = new ArrayList<Integer>();
    for (int i = 0; i < m.elementCount; i++) {
      Element e = m.elements[i];
      if (e == null || e.size <= 0) continue;
      if (compRow < e.size && compCol < e.size && e.keSet[compRow][compCol]) ids.add(i);
    }
    return ids;
  }

  void normalizeContributorFocus(ArrayList<Integer> contributorIDs) {
    if (contributorIDs == null || contributorIDs.size() == 0) {
      activeContributorIndex = 0;
      return;
    }
    activeContributorIndex = constrain(activeContributorIndex, 0, contributorIDs.size() - 1);
  }

  int getActiveContributorId(ArrayList<Integer> contributorIDs) {
    if (contributorIDs == null || contributorIDs.size() == 0) return -1;
    normalizeContributorFocus(contributorIDs);
    return contributorIDs.get(activeContributorIndex);
  }

  void render() {
    FEMDataModel m = p.model;

    viewZoom.draw(40, 80, m.elementCount);

    boolean blinkPhase = (frameCount % blinkPeriod) < blinkOn;
    int contributors = countContributors();
    ArrayList<Integer> contributorIDs = getContributorIDs();
    normalizeContributorFocus(contributorIDs);

    fill(30);
    textSize(18);
    textAlign(LEFT, CENTER);
    text("SYSTEM ARCHITECTURE ELEMENT MULTI-ARRAY FIELD TRACKER", 270, 110);

    drawContributionBanner(contributors, contributorIDs, blinkPhase);

    fill(40);
    textSize(13);
    textAlign(LEFT, CENTER);
    String state = m.locked ? "LOCKED" : (m.settingFinished ? "READY" : "EDITING");
    text("Cross Tracking: Row " + (compRow + 1) + " | Col " + (compCol + 1) +
      "  | Assembly State: " + state +
      "  | Click any cell to re-target tracker", 40, height - 150);

    textAlign(CENTER, CENTER);
    btnPrevPage.drawInner(curPage > 0);
    btnNextPage.drawInner((curPage + 1) * itemsPerPage < m.elementCount);
    btnPrevPair.drawInner(contributors > 0);
    btnNextPair.drawInner(contributors > 0);
    btnToggleLens.drawInner(true);

    int startIdx = curPage * itemsPerPage;
    int endIdx   = min(m.elementCount, startIdx + itemsPerPage);
    float spaceX   = width  * 0.31f;
    float spaceY   = height * 0.31f;
    float baseCell = 36 * viewZoom.scale;

    Element hoveredElement = null;
    int hoveredElementId = -1;
    int hoveredRow = -1, hoveredCol = -1;
    float hoveredMatrixX = 0, hoveredMatrixY = 0;

    for (int i = startIdx; i < endIdx; i++) {
      int localPos = i - startIdx;
      float matrixX = 40 + (localPos % 3) * spaceX;
      float matrixY = 160 + (localPos / 3) * spaceY;

      drawCompareCellBlock(m.elements[i], matrixX, matrixY, i, baseCell, blinkPhase, contributors, contributorIDs);

      Element e = m.elements[i];
      if (e != null && e.size > 0) {
        for (int r = 0; r < e.size; r++) {
          for (int c = 0; c < e.size; c++) {
            if (over(matrixX + c * baseCell, matrixY + r * baseCell, baseCell, baseCell)) {
              hoveredElement   = e;
              hoveredElementId = i;
              hoveredRow       = r;
              hoveredCol       = c;
              hoveredMatrixX   = matrixX;
              hoveredMatrixY   = matrixY;
            }
          }
        }
      }
    }

    if (enabledLens && hoveredElement != null) {
      executeCompareLensPipeline(
        hoveredElement, hoveredElementId,
        hoveredRow, hoveredCol, baseCell,
        hoveredMatrixX, hoveredMatrixY,
        contributors, contributorIDs, blinkPhase
      );
    }
  }

  void drawContributionBanner(int contributors, ArrayList<Integer> ids, boolean blinkPhase) {
    float bx = 270, by = 133, bw = width - bx - 40, bh = 22;

    if (contributors > 1) {
      fill(blinkPhase ? color(200, 40, 10, 200) : color(130, 20, 0, 200));
    } else if (contributors == 1) {
      fill(color(30, 130, 50, 200));
    } else {
      fill(color(90, 90, 90, 180));
    }

    noStroke();
    rect(bx, by, bw, bh, 4);

    fill(255);
    textSize(12);
    textAlign(LEFT, CENTER);

    String idStr = "";
    for (int k = 0; k < ids.size(); k++) {
      idStr += "E" + (ids.get(k) + 1);
      if (k < ids.size() - 1) idStr += (contributors > 2 && k == ids.size() - 2) ? " & " : ", ";
    }

    String focusStr = (ids.size() > 0) ? ("  |  Focus: E" + (ids.get(activeContributorIndex) + 1)) : "";

    String label;
    if (contributors > 1)      label = "► " + contributors + " ELEMENTS OVERLAP at [" + (compRow + 1) + "," + (compCol + 1) + "]  —  " + idStr + focusStr + "  ◄ MULTI-SOURCE";
    else if (contributors == 1) label = "► SINGLE CONTRIBUTION at [" + (compRow + 1) + "," + (compCol + 1) + "]  —  " + idStr + focusStr;
    else                        label = "► NO DATA at [" + (compRow + 1) + "," + (compCol + 1) + "] across any element matrix";

    text(label, bx + 10, by + bh / 2);
    textAlign(CENTER, CENTER);
  }

  void drawCompareCellBlock(Element e, float x, float y, int id, float baseCell, boolean blinkPhase, int contributors, ArrayList<Integer> contributorIDs) {
    if (e == null || e.size <= 0) return;

    boolean thisContributes = contributorIDs.contains(id);
    boolean isFocusedElement = (thisContributes && getActiveContributorId(contributorIDs) == id);
    boolean targetContributes = (compRow < e.size && compCol < e.size && e.keSet[compRow][compCol]);

    if (thisContributes && contributors > 1) {
      fill(blinkPhase ? color(255, 80, 10) : color(180, 40, 0));
    } else if (thisContributes) {
      fill(color(20, 140, 50));
    } else {
      fill(50);
    }

    textSize(12);
    textAlign(LEFT, CENTER);
    text("ELEMENT ID: " + (id + 1) + "  (size: " + e.size + ")" +
      (thisContributes && contributors > 1 ? "  ◄ CONTRIBUTES" :
       thisContributes ? "  ◄ SOLE SOURCE" : ""), x, y - 16);
    textAlign(CENTER, CENTER);

    for (int r = 0; r < e.size; r++) {
      for (int c = 0; c < e.size; c++) {
        float cx = x + c * baseCell;
        float cy = y + r * baseCell;

        boolean isTarget = (r == compRow && c == compCol);
        boolean matchRow = (r == compRow);
        boolean matchCol = (c == compCol);

          }

          cellFill = color(235, 235, 235);

        }

        fill(cellFill);

        if (isTarget && thisContributes && contributors > 1 && targetContributes) {
          stroke(blinkPhase ? color(120, 0, 0) : color(0, 80, 200));
          strokeWeight(isFocusedElement ? 3.0f : 2.2f);
        } else if (isTarget && thisContributes && targetContributes) {
          stroke(0, 100, 40);
          strokeWeight(2.0f);
        } else if (isTarget) {

          strokeWeight(2.2f);
        } else if (onCross && thisContributes && targetContributes) {
          stroke(120, 80, 40);
          strokeWeight(isFocusedElement ? 1.8f : 1.2f);
        } else if (onCross) {
          stroke(160);
          strokeWeight(1.0f);
        } else {

          strokeWeight(0.8f);

        strokeWeight(1.0f);

        if (e.keSet[r][c] && baseCell > 14) {
          fill(isTarget && thisContributes ? color(0) : color(30));
          textSize(constrain(baseCell * 0.28f, 6, 12));
          text(nf(e.ke[r][c], 0, 1), cx + baseCell / 2, cy + baseCell / 2);
        }

        if (isTarget) {
          if (thisContributes && contributors > 1 && targetContributes) {
            fill(blinkPhase ? color(80, 0, 0) : color(200, 60, 0));
          } else if (thisContributes && targetContributes) {
            fill(color(0, 120, 30));
          } else {
            fill(0);
          }
          textSize(constrain(baseCell * 0.55f, 11, 22));
          text("☝", cx + baseCell / 2, cy + baseCell / 2);

          textSize(constrain(baseCell * 0.18f, 5, 10));
          if (matchRow) text("ROW", cx + baseCell / 2, cy + 8);

      }
    }

    if (isFocusedElement) {
      noFill();
      stroke(0, 80, 200);
      strokeWeight(3.0f);
      rect(x - 4, y - 20, e.size * baseCell + 8, e.size * baseCell + 24, 8);
      strokeWeight(1.0f);
    }
  }

  boolean mousePressed() {
    FEMDataModel m = p.model;

    if (viewZoom.mousePressed()) return true;

    int contributors = countContributors();
    ArrayList<Integer> contributorIDs = getContributorIDs();
    normalizeContributorFocus(contributorIDs);

    if (btnPrevPage.hit() && curPage > 0) {
      curPage--;
      return true;
    }

    if (btnNextPage.hit() && (curPage + 1) * itemsPerPage < m.elementCount) {
      curPage++;
      return true;
    }

    if (btnPrevPair.hit() && contributors > 0) {
      if (contributorIDs.size() > 0) activeContributorIndex = (activeContributorIndex - 1 + contributorIDs.size()) % contributorIDs.size();
      return true;
    }

    if (btnNextPair.hit() && contributors > 0) {
      if (contributorIDs.size() > 0) activeContributorIndex = (activeContributorIndex + 1) % contributorIDs.size();
      return true;
    }

    if (btnToggleLens.hit()) {
      enabledLens = !enabledLens;
      return true;
    }

    float spaceX = width * 0.31f;
    float spaceY = height * 0.31f;
    float baseCell = 36 * viewZoom.scale;
    int startIdx = curPage * itemsPerPage;
    int endIdx = min(m.elementCount, startIdx + itemsPerPage);

    for (int i = startIdx; i < endIdx; i++) {
      int localPos = i - startIdx;
      float matrixX = 40 + (localPos % 3) * spaceX;
      float matrixY = 160 + (localPos / 3) * spaceY;
      Element e = m.elements[i];
      if (e == null || e.size <= 0) continue;

      for (int r = 0; r < e.size; r++) {
        for (int c = 0; c < e.size; c++) {
          if (over(matrixX + c * baseCell, matrixY + r * baseCell, baseCell, baseCell)) {
            compRow = r;
            compCol = c;

            ArrayList<Integer> newIDs = getContributorIDs();
            if (newIDs.size() == 0) activeContributorIndex = 0;
            else activeContributorIndex = min(activeContributorIndex, newIDs.size() - 1);

            int contrib = countContributors();
            p.msg = "Target locked: [" + (r + 1) + "," + (c + 1) + "]  —  " + contrib + " element(s) contribute to this position.";
            return true;
          }
        }
      }
    }

    return false;
  }

  void handleRelease() { viewZoom.stopDrag(); }
}

class ZoomControl {
  String label;
  float scale, minS, maxS, step;
  float x, y, w = 210, h = 46;
  boolean dragging = false;
  float dragOffX, dragOffY;

  ZoomControl(String lbl, float init, float mn, float mx, float stp) {
    label = lbl;
    scale = init;
    minS = mn;
    maxS = mx;
    step = stp;
  }

  void draw(float px, float py, int itemCount) {
    if (!dragging) {
      x = px;
      y = py;
    }

    fill(250);
    stroke(100);
    rect(x, y, w, h, 6);

    fill(60);
    textSize(10);
    text(label + " ZOOM  [" + nf(scale * 100, 0, 0) + "%]", x + w / 2, y + 12);

    drawSubBtn(x + 10, y + 24, 30, 16, "-");
    drawSubBtn(x + w - 40, y + 24, 30, 16, "+");

    fill(100);
    textSize(9);
    text("Tracking Nodes: " + itemCount, x + w / 2, y + 32);
  }

  void drawSubBtn(float bx, float by, float bw, float bh, String sym) {
    fill(230);
    stroke(150);
    rect(bx, by, bw, bh, 3);
    fill(0);
    textSize(10);
    text(sym, bx + bw / 2, by + bh / 2 - 1);
  }

  boolean mousePressed() {
    if (over(x + 10, y + 24, 30, 16)) {
      scale = max(minS, scale - step);
      return true;
    }
    if (over(x + w - 40, y + 24, 30, 16)) {
      scale = min(maxS, scale + step);
      return true;
    }
    if (over(x, y, w, 20)) {
      dragging = true;
      dragOffX = mouseX - x;
      dragOffY = mouseY - y;
      return true;
    }
    return over(x, y, w, h);
  }

  void drag() {
    if (dragging) {
      x = mouseX - dragOffX;
      y = mouseY - dragOffY;
    }
  }

  void stopDrag() {
    dragging = false;
  }
}

/* =========================================================
   ELEMENT MATRIX ENTITY
   ========================================================= */

class Element {
  int size = 0;
  float[][] ke;
  boolean[][] keSet;
  int[] dof;
  boolean[] dofSet;

  int asmR = 0, asmC = 0;
  boolean done = false;
  int lastR = -1, lastC = -1, lastGI = -1, lastGJ = -1;
  float lastValue = 0;

  void init(int n) {
    size = n;
    ke = new float[n][n];
    keSet = new boolean[n][n];
    dof = new int[n];
    dofSet = new boolean[n];
    resetAssemblyProgress();
  }

  void resetAll() {
    size = 0;
    ke = null;
    keSet = null;
    dof = null;
    dofSet = null;
    resetAssemblyProgress();
  }

  void resetAssemblyProgress() {
    asmR = 0;
    asmC = 0;
    done = false;
    lastR = -1;
    lastC = -1;
    lastGI = -1;
    lastGJ = -1;
    lastValue = 0;
  }

  void autoMatrix() {
    if (size <= 0) return;
    for (int r = 0; r < size; r++) {
      for (int c = 0; c < size; c++) {
        ke[r][c] = random(-500, 1000);
        keSet[r][c] = true;
      }
    }
    resetAssemblyProgress();
  }

  void autoDOF(int gstN) {
    if (size <= 0 || gstN < size) return;
    ArrayList<Integer> pool = new ArrayList<Integer>();
    for (int i = 0; i < gstN; i++) pool.add(i + 1);

    for (int i = 0; i < size; i++) {
      int rIdx = int(random(pool.size()));
      dof[i] = pool.remove(rIdx) - 1;
      dofSet[i] = true;
    }
    resetAssemblyProgress();
  }

  void autoBandDOF(int gstN, int startDOF) {
    if (size <= 0 || gstN < size) return;
    int start = constrain(startDOF, 1, gstN - size + 1);
    for (int i = 0; i < size; i++) {
      dof[i] = (start - 1) + i;
      dofSet[i] = true;
    }
    resetAssemblyProgress();
  }

  boolean matrixComplete() {
    if (size == 0) return false;
    for (int r = 0; r < size; r++) {
      for (int c = 0; c < size; c++) {
        if (!keSet[r][c]) return false;
      }
    }
    return true;
  }

  boolean dofComplete() {
    if (size == 0) return false;
    for (int i = 0; i < size; i++) if (!dofSet[i]) return false;
    return true;
  }

  boolean complete() {
    return size > 0 && matrixComplete() && dofComplete();
  }

  boolean readyForAssembly() {
    return complete() && !done;
  }

  boolean stepOne(GlobalMatrix G) {
    if (!readyForAssembly()) return false;

    lastR = asmR;
    lastC = asmC;
    lastGI = dof[asmR];
    lastGJ = dof[asmC];
    lastValue = ke[asmR][asmC];
    G.add(lastGI, lastGJ, lastValue);

    asmC++;
    if (asmC >= size) {
      asmC = 0;
      asmR++;
    }
    if (asmR >= size) done = true;
    return true;
  }

  void drawLocal(float x, float y, float cell, Keypad kp, boolean hiAsm, float zoom, int actId) {
    pushStyle();
    fill(0);
    textSize(14);
    textAlign(LEFT, CENTER);
    text("LOCAL STIFFNESS COMPONENT: ELEMENT " + (actId + 1), x, y - 20);
    textAlign(CENTER, CENTER);

    float dCell = cell * zoom;
    textSize(constrain(dCell * 0.32f, 6, 13));

    for (int r = 0; r < size; r++) {
      for (int c = 0; c < size; c++) {
        boolean sel = (kp.mode == INPUT_MATRIX && kp.el == this && kp.r == r && kp.c == c);

        if      (hiAsm && r == lastR && c == lastC) fill(255, 150, 150);
        else if (sel)                               fill(255, 230, 140);
        else                                        fill(keSet[r][c] ? 240 : 255);

        stroke(50);
        rect(x + c * dCell, y + r * dCell, dCell, dCell);

        fill(0);
        String t = "";
        if (keSet[r][c]) t = nf(ke[r][c], 0, 1);
        if (sel && kp.val.length() > 0) t = kp.val;
        text(t, x + c * dCell + dCell / 2, y + r * dCell + dCell / 2);
      }
    }
    popStyle();
  }

  void drawDOF(float x, float y, float cell, Keypad kp, boolean hiAsm, float zoom) {
    pushStyle();
    fill(0);
    textSize(13);
    textAlign(LEFT, CENTER);
    text("SYSTEM NODE MAP TRANSLATIONS (Degrees of Freedom Boundary Coordinates)", x, y - 16);
    textAlign(CENTER, CENTER);

    float dCell = cell * zoom;
    textSize(constrain(dCell * 0.34f, 6, 13));

    for (int i = 0; i < size; i++) {
      boolean sel = (kp.mode == INPUT_DOF && kp.index == i);

      if      (hiAsm && (i == lastR || i == lastC)) fill(255, 150, 150);
      else if (sel)                                 fill(255, 230, 140);
      else                                          fill(dofSet[i] ? 240 : 255);

      stroke(50);
      rect(x + i * dCell, y, dCell, dCell);

      fill(0);
      String t = "";
      if (dofSet[i]) t = "g" + (dof[i] + 1);
      if (sel && kp.val.length() > 0) t = kp.val;
      text(t, x + i * dCell + dCell / 2, y + dCell / 2);
    }
    popStyle();
  }
}

/* =========================================================
   GLOBAL MATRIX
   ========================================================= */

class GlobalMatrix {
  int N = 0;
  float[][] K;
  int[][] records;
  int hiI = -1, hiJ = -1, hiElem = -1;
  float hiV = 0;

  GlobalMatrix(int max) {
    K = new float[max][max];
    records = new int[max][max];
  }

  void setSize(int n) {
    this.N = n;
    reset();
  }

  void reset() {
    for (int i = 0; i < K.length; i++) {
      for (int j = 0; j < K.length; j++) {
        K[i][j] = 0;
        records[i][j] = 0;
      }
    }
    clearHighlight();
  }

  void clearHighlight() {
    hiI = -1;
    hiJ = -1;
    hiElem = -1;
    hiV = 0;
  }

  void setHighlight(int I, int J, float v, int e) {
    hiI = I;
    hiJ = J;
    hiV = v;
    hiElem = e;
  }

  void add(int I, int J, float val) {
    if (I >= 0 && J >= 0 && I < N && J < N) {
      K[I][J] += val;
      records[I][J]++;
    }
  }

  void draw(float x, float y, float cell, float zoom) {
    pushStyle();
    fill(0);
    textSize(14);
    textAlign(LEFT, CENTER);
    text("GLOBAL STIFFNESS SYSTEM ARRAY (" + N + " x " + N + ")", x, y - 20);
    textAlign(CENTER, CENTER);

    float dCell = cell * zoom;
    textSize(constrain(dCell * 0.32f, 5, 12));

    for (int r = 0; r < N; r++) {
      for (int c = 0; c < N; c++) {
        if      (r == hiI && c == hiJ) fill(255, 150, 150);
        else if (records[r][c] > 0)    fill(160, 220, 180);
        else                           fill(255);

        stroke(160);
        rect(x + c * dCell, y + r * dCell, dCell, dCell);

        if (dCell > 10 || records[r][c] > 0) {
          fill(0);
          text(nf(K[r][c], 0, 1), x + c * dCell + dCell / 2, y + r * dCell + dCell / 2);
        }
      }
    }
    popStyle();
  }
}

/* =========================================================
   KEYPAD
   ========================================================= */

class Keypad {
  FEMApplication app;
  String val = "";
  int mode = INPUT_NONE;
  Element el;
  int r, c, index;

  float x = 280, y = 140, w = 250, h = 280;
  boolean dragging = false;
  float offX, offY;

  String[] gridLayout = {
    "7","8","9",
    "4","5","6",
    "1","2","3",
    "0",".","-",
    "⌫","C","OK"
  };

  Keypad(FEMApplication app) {
    this.app = app;
  }

  void startGST() {
    mode = INPUT_GST;
    val = "";
    el = null;
  }

  void startElementCount() {
    mode = INPUT_ECOUNT;
    val = "";
    el = null;
  }

  void startSize(Element e) {
    mode = INPUT_SIZE;
    el = e;
    val = (e.size > 0) ? str(e.size) : "";
  }

  void startMatrix(Element e, int rr, int cc) {
    mode = INPUT_MATRIX;
    el = e;
    r = rr;
    c = cc;
    val = e.keSet[rr][cc] ? str(e.ke[rr][cc]) : "";
  }

  void startDOF(Element e, int idx) {
    mode = INPUT_DOF;
    el = e;
    index = idx;
    val = e.dofSet[idx] ? str(e.dof[idx] + 1) : "";
  }

  String getLabel() {
    if (mode == INPUT_GST)    return "INITIALIZE SYSTEM DIMENSION";
    if (mode == INPUT_ECOUNT) return "TOTAL DISCRETE COMPONENT QUANTITY";
    if (mode == INPUT_SIZE)   return "ELEMENT ARRAY SIZE VECTOR";
    if (mode == INPUT_MATRIX)  return "COEFFICIENT (" + (r + 1) + "," + (c + 1) + ")";
    if (mode == INPUT_DOF)     return "NODE TRANSLATION ID POOL POSITION (" + (index + 1) + ")";
    return "";
  }

  void draw() {
    if (mode == INPUT_NONE) return;

    fill(255);
    stroke(0);
    strokeWeight(1.5f);
    rect(x, y, w, h, 10);
    strokeWeight(1.0f);

    fill(230);
    rect(x, y, w, 32, 10, 10, 0, 0);
    fill(0);
    textSize(10);
    text(getLabel(), x + w / 2, y + 16);

    for (int i = 0; i < gridLayout.length; i++) {
      float bx = x + 12 + (i % 3) * 76;
      float by = y + 42 + (i / 3) * 42;
      fill(240);
      stroke(100);
      rect(bx, by, 70, 36, 4);
      fill(0);
      textSize(12);
      text(gridLayout[i], bx + 35, by + 18);
    }

    fill(255);
    stroke(100);
    rect(x + 12, y + h - 32, w - 24, 24, 4);
    fill(val.length() > 0 ? 0 : 150);
    textSize(11);
    text(val.length() > 0 ? val : "Await Character Map Submission...", x + w / 2, y + h - 20);
  }

  boolean mousePressed() {
    if (mode == INPUT_NONE) return false;

    if (over(x, y, w, 32)) {
      dragging = true;
      offX = mouseX - x;
      offY = mouseY - y;
      return true;
    }

    for (int i = 0; i < gridLayout.length; i++) {
      float bx = x + 12 + (i % 3) * 76;
      float by = y + 42 + (i / 3) * 42;
      if (over(bx, by, 70, 36)) {
        processKeyStroke(gridLayout[i]);
        return true;
      }
    }

    return over(x, y, w, h);
  }

  void processKeyStroke(String key) {
    if      (key.equals("C"))  val = "";
    else if (key.equals("⌫"))  { if (val.length() > 0) val = val.substring(0, val.length() - 1); }
    else if (key.equals("OK"))  { commit(); }
    else {
      if (mode == INPUT_MATRIX && (key.equals(".") || key.equals("-"))) val += key;
      else if (Character.isDigit(key.charAt(0))) val += key;
    }
  }

  void commit() {
    FEMDataModel m = app.model;
    if (val.length() == 0) return;

    if (mode == INPUT_GST) {
      if (!isPositiveIntString(val)) return;
      m.gstN = constrain(int(val), 1, MAX_G);
      m.gstSet = true;
      m.G.setSize(m.gstN);
      m.invalidateAssemblyState();
      m.elementCount = 0;
      mode = INPUT_ECOUNT;
      app.msg = "Global node boundaries set. Enter structural element count.";
      val = "";
      return;
    }

    if (mode == INPUT_ECOUNT) {
      if (!isPositiveIntString(val)) return;
      m.elementCount = constrain(int(val), 1, MAX_E);
      m.invalidateAssemblyState();
      for (int i = 0; i < MAX_E; i++) m.elements[i].resetAll();
      m.active = 0;
      mode = INPUT_SIZE;
      el = m.elements[0];
      app.msg = "Stiffness elements structured. Setup parameters for Element 1.";
      val = "";
      return;
    }

    if (mode == INPUT_SIZE) {
      if (!isPositiveIntString(val)) return;
      int s = constrain(int(val), 1, m.gstN);
      el.init(s);
      m.invalidateAssemblyState();
      startMatrix(el, 0, 0);
      app.msg = "Internal bounds loaded. Input processing cell constants.";
      return;
    }

    if (mode == INPUT_MATRIX) {
      if (!isFloatString(val)) return;
      el.ke[r][c] = float(val);
      el.keSet[r][c] = true;
      m.invalidateAssemblyState();

      int total = el.size * el.size;
      int next = r * el.size + c + 1;
      if (next < total) {
        startMatrix(el, next / el.size, next % el.size);
      } else {
        startDOF(el, 0);
        app.msg = "Cell configurations complete. Bind systemic target boundary vector paths.";
      }
      return;
    }

    if (mode == INPUT_DOF) {
      if (!isPositiveIntString(val)) return;
      int d = constrain(int(val), 1, m.gstN);
      el.dof[index] = d - 1;
      el.dofSet[index] = true;
      m.invalidateAssemblyState();

      if (index + 1 < el.size) {
        startDOF(el, index + 1);
      } else {
        el.resetAssemblyProgress();
        if (m.active < m.elementCount - 1) {
          m.active++;
          el = m.elements[m.active];
          mode = INPUT_SIZE;
          app.msg = "Element validation matrix structured. Define dimensional scale for Component: " + (m.active + 1);
        } else {
          app.msg = "All elements mapped. Execute layout compiling with 'FINISH'.";
          mode = INPUT_NONE;
        }
      }
    }

    val = "";
  }

  void drag() {
    if (dragging) {
      x = mouseX - offX;
      y = mouseY - offY;
    }
  }

  void stopDrag() {
    dragging = false;
  }
}
